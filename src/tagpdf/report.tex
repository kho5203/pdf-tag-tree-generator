% A vertically merged data cell can refer to a row header that comes later.
% Save each cell's structure number now, then resolve header IDs at table end.
\seq_new:N \g_tagpdf_table_headers_seq
\NewDocumentCommand\TagApplyHeaders{m}{
  \seq_gput_right:Ne \g_tagpdf_table_headers_seq {
    \exp_not:N \tag_struct_gput:nne {\tag_get:n{struct_num}}{attribute}
      {/O~/Table~/Headers~[\exp_not:n{#1}]}
  }
}
\NewDocumentCommand\TagFinishTable{}{
  \seq_map_function:NN \g_tagpdf_table_headers_seq \use:n
  \seq_gclear:N \g_tagpdf_table_headers_seq
}
